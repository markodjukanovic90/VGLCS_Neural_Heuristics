\begin{frame}{Introduction}
   \begin{itemize}
       \item Objects we deal with: sequences (strings) over finite alphabet  $\Sigma$
       \begin{itemize}
       \item DNA/RNA sequences over $\{\texttt{A}, \texttt{T}, \texttt{G}, \texttt{C/U}\}$
       \item Proteins over 20 (canonical) amino acids: $\{\texttt{A}, \texttt{C}, \texttt{D}, \texttt{E}, \texttt{P}, \texttt{Q} ...\}$ 
       \end{itemize} \vspace{0.3cm}
        \item \textbf{One of central tasks in computational biology:} 
        \begin{itemize}
        	\item  sequence comparison, find \textbf{common motifs} between sequences
          \item structural similarity vs. sequence functionality 
       \end{itemize} \vspace{0.4cm}
       
       \item Motifs = common \textbf{subsequences} $\rightarrow$ \textcolor{blue}{\textbf{Longest commmon subsequence problem}} variants (studied 50 years already)
   \end{itemize}
\end{frame}

\begin{frame}{Longest common subsequence problem (LCSP)}
   
    
    \begin{definition}[LCSP]
        \textbf{Input}: Given an arbitrary set of sequences $S=\{s_1, \ldots, s_m\}$ \\
        \textbf{Task}: Find a longest possible  subsequence $s$  \textcolor{blue}{common} for all sequences from $S$. %with \textcolor{blue}{maximum} possible length ($|s|$). 
    \end{definition} \vspace{0.5cm}
   % \begin{example}
    %   Input: $S=$\{\texttt{\textcolor{red}{A}A\textcolor{red}{T}\textcolor{red}{T}G\textcolor{red}{C}}, \texttt{\textcolor{red}{A}\textcolor{red}{T}\textcolor{red}{T}A\textcolor{red}{C}}\} \\ 
    %   LCS solution: $s=\text{\textcolor{red}{ATTC}}$
    %\end{example}
    %   \vspace{0.3cm}
  
 \textbf{LCS: Literature \& Problem Variants }
  \begin{itemize}
  \item For $m=2$: polynomially solvable, e.g. {Dynamic programming} %Hunt-Szymanski, Hirschenberg,  ... 
 \vspace{0.2cm}
  \item When $m$ arbitrary large:   \textcolor{red}{$\mathcal{NP}$-hard}, e.g.,  approximation, meta-heuristic, exact approaches   
  \end{itemize}\vspace{0.5cm}
  
  \textbf{Problem-related practical variants}:    Repetition-free, Filled LCS problem, \textcolor{red}{Gapped LCS problem}
\end{frame}

\begin{frame}{The gapped LCS problem (Peng and Yang, 2012)}
   \begin{definition}[Gapped sequence]
       Given is a sequence $s$ and a function 
       $G_s \colon \{1, \ldots, |s|\} \mapsto \mathbb{N}$. A pair \textcolor{red}{$(s, G_s)$} is called a {gapped sequence}.
   \end{definition} ($G_s$: one number assigned  to each position of $s$) \vspace{0.5cm}
   
   \begin{definition}[Gapped subsequence]
         Sequence $\tilde{s}$ is a \textcolor{red}{gapped subsequence} of $(s, G_s)$ iff 
         \begin{itemize}
             \item $\tilde{s}$ is a \textbf{subsequence} of $s$
             \item the gap constraint $G_s$ fulfilled resp. the \textit{positions of embedding} $\tilde{s}$ in $s$:
         \begin{itemize}
         	\item \textcolor{red}{upper bounds assigned on the number of letters} between  consecutive pairs of (embedding) positions
             %\item  suppose $i_1< \ldots <i_{|\tilde{s}|}$ are those \textbf{positions}  
             %\item ($\forall$ $j=2, \ldots, |\tilde{s}|)\, \textcolor{red}{i_{j} - i_{j-1} \leq G_{s}(i_{j})+1}$  (consecutive positions respect gap distances = allowed \textit{number of letters inbetween})
         \end{itemize}
       \end{itemize}
   \end{definition}
   
\end{frame}

\begin{frame}{Problem definition}

   \begin{example}
      $s=\texttt{AATTGC}$, $\textcolor{red}{G_s(\cdot)=1}$, $\tilde{s}=\texttt{ATG}$
      \begin{itemize}
         \item the embedding \textcolor{red}{A}A\textcolor{red}{T}T\textcolor{red}{G}C (\textcolor{blue}{valid} gapped subsequence)
         \item  the embedding \textcolor{red}{A}A{T}\textcolor{red}{T}\textcolor{red}{G}C (\textcolor{blue}{invalid}, two letter inbetween $> G_s[4]=1$)
      \end{itemize}
   \end{example} \vspace{0.5cm}
    
   \begin{definition}[The multiple gapped LCS problem]
        \textbf{Input}: Given is an arbitrary set of  gapped sequences $\{ (s_i, G_{s_i})\}_{i=1}^m \}$. \\
        \textbf{Task}: Find the longest  sequence $\tilde{s}$ so that 
        \begin{itemize}
            \item  $\tilde{s}$ is \textcolor{red}{\textbf{common subusequence}} of each $s_i$
            \item $\tilde{s}$ \textcolor{red}{\textbf{fulfills all gap constraints}} $G_{s_i}$ ($i=1, \ldots, m$)
        \end{itemize}
    \end{definition}
   % \textbf{Note}: When $G_{s_i}=n$ (length of longest sequence) $\Rightarrow$ VGLCSP = LCSP.
\end{frame}

\begin{frame}{Literature \& Motivation for VGLCSP}
   
   \begin{itemize}
       \item  Peng and Yang (2012, 2014):  $m=2$  version by \textcolor{red}{three dynamic programming} (DP) approaches 
       \item  Manea et al. (2024): complexity analysis investigated
       \begin{itemize}
       	\item   {\textcolor{red}$\mathcal{NP}$-hard }  ($m$ arbitrary) 
        \end{itemize} 
        \item Djukanovic et al. (2026) the first approach for $m$ arbitrary: \textcolor{red}{Iterative Beam search approach}
   \end{itemize} \vspace{0.5cm}
   
   \textbf{Motivation}:
   \begin{itemize}
       \item  \textcolor{blue}{Genetics and molecular biology}%: %DNA/protein comparison analysis--distances between residues must be respected \vspace{0.2cm}
       \item \textcolor{blue}{Time-series analysis}: in settings where events are required to occur within specified temporal delays~(Lainscsek et al. (2015))
   \end{itemize}
   
\end{frame}

\begin{frame}{Methodology by Djukanovic et al. (2026) }
	
%	STARTING FROM HERE NEEDS TO BE ADAPTED...
	
    \begin{itemize}
        \item Based on the \textbf{state space graph formulation} of the problem \vspace{0.3cm}
        \item \textbf{Nodes} corresp. to partial solutions and resp. subproblems to solve
        \begin{itemize}
        	\item transitions: valid extensions of partial solutions (by one letter)
        \end{itemize} \vspace{0.3cm}
        \item \textbf{Gap constraints}: cut-off invalid extensions (edges) among the LCS extensions \vspace{0.3cm}
        \item \textcolor{red}{\textbf{Many root/source nodes}} in the state space%---the position of leading letter in each common gapped subsequence free to select 
        \begin{itemize}
        	\item   root state space subgraph (formulation) w.r.t. starting source (match)
        \end{itemize} 
    \end{itemize}
    
\end{frame}

  \begin{itemize}
      
      %\item Start with $p^L=1$, and empty partial solution, refers to the complete problem
      
      \item Each \textbf{state} $v=(p^L, l^v)$: one or more feasible partial solutions where
      \begin{itemize}
      	  \item  a vector of positions $p^L$ refer to the positions of suffixes of input sequences relevant to  further expand these sols (subproblem)
      	  \item the  length of current partial solution $l^v$ 
      \end{itemize}\vspace{0.3cm}
      \item \textbf{Expansion} of $v$: extend (concatenate)  partial solutions feasibly (and non-dominantly) by one letter  in all possible ways,  respecting gap constraints \vspace{0.3cm}
      \item \textbf{Non-expandable nodes}: complete solutions; 
  \end{itemize} \vspace{0.15cm}
  
  \fbox{\textcolor{red}{\textbf{Decision: select appropriate (matches)\ $p^L$ for start (empty solution)}}} $\Rightarrow$ possibly \textbf{(exponentially) many root nodes}  \\ \vspace{0.2cm} {Space($p^L$)}: root-based state (sub) graph  induced by root node $(p^L, 0)$.
\end{frame}

\begin{frame}{Methodology: main components roughly}%{An issue with the root-state-space formulation: disconnected components}
	
	\begin{itemize}
		\item  Generation of high-quality  solutions for a root (sub-)space generally hard $\Rightarrow$  an approx. approach for evaluation (\textcolor{red}{\textbf{Beam search} -- \textbf{limited BFS}}) \vspace{0.5cm}
		\item  The root-state-space formulation may involve disconnected components  $\Rightarrow$ needs \textbf{iterative beam search} to explore all search space 
	\end{itemize} \vspace{0.7cm}
	
	$\Longrightarrow$ \textcolor{red}{\textbf{ Iterative Multi Source Beam Search (IMSBS) approach}}. 
\end{frame}

\begin{frame}{Learning weight of NN evaluator: Why not standard supervised backpropagation?}
\begin{columns}[T,totalwidth=\textwidth]

\begin{block}{Obstacle(s)}
\begin{itemize}
  %\item No direct target labels for ``good node'' on large VGLCSP instances

\begin{column}{0.46\textwidth}
\begin{block}{Solution: neuro-evolution}
\begin{itemize}
  \item Treat all NN weights as \textbf{one vector $w$}.
  \item Evaluate $w$ by running (inexpensive) IMSBS on training (32) instances.
  \item \textbf{Fitness} = average subsequence length found
\end{itemize}
\end{block}
